WITH params AS (
  SELECT
    '{{ ListeAffaires.selectedRow.numero_affaire + InputRepereAppareil.text }}'::text AS num_app,
    '{{ ListeAffaires.selectedRow.numero_affaire }}'::text AS num_aff,
    '{{ InputNomAppareil.text }}'::text AS nom_appareil,
    '{{ InputRepereAppareil.text }}'::text AS repere,
    {{ DateDelai.selectedDate ? moment(DateDelai.selectedDate).format("YYYY-MM-DD") : null }}::date AS delai,
    '{{ appsmith.store.appareil_source.numero_appareil }}'::text AS num_app_src
),

ins_app AS (
  INSERT INTO public.appareils (
    numero_appareil,
    numero_affaire,
    delai,
    cloture_suivi_fabrication
  )
  SELECT
    p.num_app,
    p.num_aff,
    p.delai,
    false
  FROM params p
  RETURNING numero_appareil
),

old_groups AS (
  SELECT
    g.id AS old_groupe_appareil_id,
    g.groupe_id,
    g.responsable,
    g.statut_id
  FROM public.workflow_groupe_appareil g
  JOIN params p ON TRUE
  WHERE g.numero_appareil = p.num_app_src
),

ins_groups AS (
  INSERT INTO public.workflow_groupe_appareil (
    numero_appareil,
    groupe_id,
    responsable,
    statut_id
  )
  SELECT
    p.num_app,
    og.groupe_id,
    og.responsable,
    og.statut_id
  FROM old_groups og
  JOIN params p ON TRUE
  RETURNING id AS new_groupe_appareil_id, groupe_id
),

group_map AS (
  SELECT
    og.old_groupe_appareil_id,
    ig.new_groupe_appareil_id,
    og.groupe_id
  FROM old_groups og
  JOIN ins_groups ig ON ig.groupe_id = og.groupe_id
),
